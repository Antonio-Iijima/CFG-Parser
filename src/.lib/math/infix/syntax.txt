<MATHEXPR>   ::= <ADD> | <SUBTRACT> | <TERM> | <COMPARISON>
<ADD>        ::= <MATHEXPR> \+ <TERM>
<SUBTRACT>   ::= <MATHEXPR> - <TERM>
<TERM>       ::= <MULTIPLY> | <DIVIDE> | <FACTOR>
<MULTIPLY>   ::= <TERM> \* <FACTOR>
<DIVIDE>     ::= <TERM> / <FACTOR>
<FACTOR>     ::= \( <MATHEXPR> \) | <VALUE>
<VALUE>      ::= <NUMBER>

<COMPARISON> ::= <LE> | <LEQ> | <GE> | <GEQ> | <EQ> | <NEQ>
<LE>         ::= <MATHEXPR> < <COMPARISON>
<LEQ>        ::= <MATHEXPR> <= <COMPARISON>
<GE>         ::= <MATHEXPR> > <COMPARISON>
<GEQ>        ::= <MATHEXPR> >= <COMPARISON>
<EQ>         ::= <MATHEXPR> == <COMPARISON>
<NEQ>        ::= <MATHEXPR> != <COMPARISON>
