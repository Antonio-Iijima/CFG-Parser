#require datatypes.identifier
#require datatypes.numeric
#require datatypes.string
#require math.infix
#require stdlib


PROGRAM          -> STATEMENTS | EXPRESSION

STATEMENTS       -> STATEMENT | STATEMENT STDLIB.NEWLINE STATEMENTS
STATEMENT        -> ASSIGNMENT | CONDITIONAL | RETURN | GOTO | MARKER | PRINT

ASSIGNMENT       -> "let" DATATYPES.IDENTIFIER.ID "be" EXPRESSION

CONDITIONAL      -> IF_THEN | IF_THEN_ELSE
IF_THEN          -> "if" EXPRESSION "," "then" BLOCK
IF_THEN_ELSE     -> "if" EXPRESSION "," "then" BLOCK STDLIB.NEWLINE "else" BLOCK
BLOCK            -> STDLIB.NEWLINE INDENT STATEMENTS DEDENT | STATEMENT

RETURN           -> "return" EXPRESSION

GOTO             -> "goto" "instruction" MARK
MARKER           -> "@" MARK
MARK             -> DATATYPES.IDENTIFIER.ID | DATATYPES.NUMERIC.NUMBER

PRINT            -> "print" EXPRESSION | "print"

EXPRESSION       -> MATH.INFIX.INFIX STDLIB.COMMENT | MATH.INFIX.INFIX
MATH.INFIX.VALUE -> DATATYPES.STRING.STRING | LABEL
LABEL            -> DATATYPES.IDENTIFIER.ID
