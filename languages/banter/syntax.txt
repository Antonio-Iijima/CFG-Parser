#require datatypes.identifier
#require datatypes.string
#require math.infix
#require stdlib


PROGRAM      -> STATEMENTS | EXPRESSION

STATEMENTS   -> STATEMENT | STATEMENT NEWLINE STATEMENTS
STATEMENT    -> ASSIGNMENT | CONDITIONAL | RETURN | GOTO | MARKER | PRINT

ASSIGNMENT   -> "let" ID "be" EXPRESSION

CONDITIONAL  -> IF_THEN | IF_THEN_ELSE
IF_THEN      -> "if" EXPRESSION "," "then" BLOCK
IF_THEN_ELSE -> "if" EXPRESSION "," "then" BLOCK NEWLINE "else" BLOCK
BLOCK        -> NEWLINE INDENT STATEMENTS DEDENT

RETURN       -> "return" EXPRESSION

GOTO         -> "goto" "instruction" MARK
MARKER       -> "@" MARK
MARK         -> ID | NUMBER

PRINT        -> "print" EXPRESSION | "print"

EXPRESSION   -> MATHEXPR COMMENT | MATHEXPR
VALUE        -> STRING | LABEL
LABEL        -> ID
