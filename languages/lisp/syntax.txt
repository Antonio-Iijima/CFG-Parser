#require datatypes.numeric
#require datatypes.string


PROGRAM   -> LISP_EXPR | LISP_EXPR PROGRAM
LISP_EXPR -> "\'" S_EXPR | S_EXPR
S_EXPR    -> ATOM | LIST

ATOM      -> NUMBER | STRING | OPERATOR | "[_a-zA-Z][^\s\'()]*"

LIST      -> "\(" "\)" | "\(" ELEMS "\)"
ELEMS     -> ELEM | ELEM ELEMS
ELEM      -> LISP_EXPR

OPERATOR  -> "[+*/<>%?]" | "\-" | "//" | "\*\*" | "<=" | ">=" | "=="
